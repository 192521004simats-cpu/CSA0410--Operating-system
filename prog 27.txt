#include <stdio.h>

void optimal_page_replacement(int pages[], int n, int capacity) {
    int frames[capacity];
    int page_faults = 0;

    for (int i = 0; i < capacity; i++) frames[i] = -1;

    printf("\nEvent\t\tFrame State\n-----------------------------\n");
    for (int i = 0; i < n; i++) {
        int available = 0;

        for (int j = 0; j < capacity; j++) {
            if (frames[j] == pages[i]) {
                available = 1;
                break;
            }
        }

        if (available == 0) {
            int replaced_idx = -1;

            for (int j = 0; j < capacity; j++) {
                if (frames[j] == -1) {
                    replaced_idx = j;
                    break;
                }
            }

            if (replaced_idx == -1) {
                int farthest = i;
                for (int j = 0; j < capacity; j++) {
                    int k;
                    for (k = i + 1; k < n; k++) {
                        if (frames[j] == pages[k]) {
                            if (k > farthest) {
                                farthest = k;
                                replaced_idx = j;
                            }
                            break;
                        }
                    }
                    if (k == n) { 
                        replaced_idx = j;
                        break;
                    }
                }
            }

            frames[replaced_idx] = pages[i];
            page_faults++;
            printf("Miss (%d)\t", pages[i]);
        } else {
            printf("Hit  (%d)\t", pages[i]);
        }

        for (int j = 0; j < capacity; j++) {
            if (frames[j] != -1) printf("%d ", frames[j]);
            else printf("- ");
        }
        printf("\n");
    }
    printf("\nTotal Page Faults: %d\n", page_faults);
}

int main() {
    int pages[] = {7, 0, 1, 2, 0, 3, 0, 4, 2, 3, 0, 3, 2};
    int n = sizeof(pages) / sizeof(pages[0]);
    int capacity = 3;

    optimal_page_replacement(pages, n, capacity);
    return 0;
}
