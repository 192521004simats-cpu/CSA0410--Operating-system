#include <stdio.h>
#include <stdlib.h>
#include <sys/ipc.h>
#include <sys/msg.h>
#include <unistd.h>
#include <sys/wait.h>

struct message {
    long type;
    char text[100];
};

int main() {
    int id;
    struct message msg;

    id = msgget(IPC_PRIVATE, 0666 | IPC_CREAT);

    if (fork() == 0) {
        msgrcv(id, &msg, sizeof(msg.text), 1, 0);
        printf("Child received: %s\n", msg.text);
    } else {
        msg.type = 1;
        sprintf(msg.text, "Hello from Parent");
        msgsnd(id, &msg, sizeof(msg.text), 0);
        wait(NULL);
        msgctl(id, IPC_RMID, NULL);
    }

    return 0;
}
