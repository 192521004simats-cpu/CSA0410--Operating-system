#include <stdio.h>
#include <string.h>

int main() {
    char files[10][20];
    int n = 0, choice;
    char name[20];

    while (1) {
        printf("\n1.Create  2.Delete  3.Search  4.Display  5.Exit\n");
        printf("Enter choice: ");
        scanf("%d", &choice);

        if (choice == 1) {
            printf("Enter file name: ");
            scanf("%s", name);

            strcpy(files[n++], name);
            printf("File created\n");
        }
        else if (choice == 2) {
            printf("Enter file name: ");
            scanf("%s", name);

            for (int i = 0; i < n; i++) {
                if (strcmp(files[i], name) == 0) {
                    strcpy(files[i], files[n - 1]);
                    n--;
                    printf("File deleted\n");
                    break;
                }
            }
        }
        else if (choice == 3) {
            printf("Enter file name: ");
            scanf("%s", name);

            for (int i = 0; i < n; i++)
                if (strcmp(files[i], name) == 0) {
                    printf("File found\n");
                    return 0;
                }

            printf("File not found\n");
        }
        else if (choice == 4) {
            printf("Files:\n");
            for (int i = 0; i < n; i++)
                printf("%s\n", files[i]);
        }
        else {
            break;
        }
    }

    return 0;
}
