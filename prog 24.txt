#include <stdio.h>
#include <fcntl.h>
#include <unistd.h>
#include <string.h>

int main() {
    int fd;
    char write_buf[] = "Hello, exploring UNIX File System Management Calls.";
    char read_buf[100];

    fd = open("unix_demo.txt", O_CREAT | O_WRONLY | O_TRUNC, 0644);
    if (fd < 0) {
        perror("Error creating file");
        return 1;
    }
    printf("File created and opened successfully. FD: %d\n", fd);

    // 2. WRITE data to the file
    int bytes_written = write(fd, write_buf, strlen(write_buf));
    if (bytes_written < 0) {
        perror("Error writing to file");
        close(fd);
        return 1;
    }
    printf("Successfully wrote %d bytes to the file.\n", bytes_written);
    close(fd); 

    fd = open("unix_demo.txt", O_RDONLY);
    if (fd < 0) {
        perror("Error opening file for reading");
        return 1;
    }

    int bytes_read = read(fd, read_buf, sizeof(read_buf) - 1);
    if (bytes_read < 0) {
        perror("Error reading from file");
        close(fd);
        return 1;
    }
    read_buf[bytes_read] = '\0'; 
    printf("Successfully read %d bytes. Content:\n\"%s\"\n", bytes_read, read_buf);
    close(fd); 

    if (unlink("unix_demo.txt") == 0) {
        printf("Demo file successfully unlinked (deleted) from system.\n");
    } else {
        perror("Error deleting file");
    }

    return 0;
}
