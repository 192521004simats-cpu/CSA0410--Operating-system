#include <stdio.h>

void lru_page_replacement(int pages[], int n, int capacity) {
    int frames[capacity], time[capacity];
    int page_faults = 0, counter = 0;

    for (int i = 0; i < capacity; i++) frames[i] = -1;

    printf("\nEvent\t\tFrame State\n-----------------------------\n");
    for (int i = 0; i < n; i++) {
        int available = 0;

        for (int j = 0; j < capacity; j++) {
            if (frames[j] == pages[i]) {
                available = 1;
                counter++;
                time[j] = counter; 
                break;
            }
        }

        if (available == 0) {
            int least_used_idx = 0;
            for (int j = 1; j < capacity; j++) {
                if (frames[j] == -1) {
                    least_used_idx = j;
                    break;
                }
                if (time[j] < time[least_used_idx]) {
                    least_used_idx = j;
                }
            }

            frames[least_used_idx] = pages[i];
            counter++;
            time[least_used_idx] = counter;
            page_faults++;
            printf("Miss (%d)\t", pages[i]);
        } else {
            printf("Hit  (%d)\t", pages[i]);
        }

        for (int j = 0; j < capacity; j++) {
            if (frames[j] != -1) printf("%d ", frames[j]);
            else printf("- ");
        }
        printf("\n");
    }
    printf("\nTotal Page Faults: %d\n", page_faults);
}

int main() {
    int pages[] = {7, 0, 1, 2, 0, 3, 0, 4, 2, 3, 0, 3, 2};
    int n = sizeof(pages) / sizeof(pages[0]);
    int capacity = 3;

    lru_page_replacement(pages, n, capacity);
    return 0;
}
