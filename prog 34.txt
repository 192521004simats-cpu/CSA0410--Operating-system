#include <stdio.h>
#include <stdlib.h>
#include <string.h>

void display_permissions_info();
void parse_symbolic_notation(const char *symbolic);
void parse_octal_notation(int octal);
void clear_input_buffer();

int main() {
    int choice;
    char symbolic_input[11];
    int octal_input;

    while (1) {
        printf("\n==================================================\n");
        printf("  QUESTION 34: LINUX FILE ACCESS PERMISSIONS SIMULATOR  \n");
        printf("==================================================\n");
        printf("1. View User Types and Permission Theory Layout\n");
        printf("2. Convert Symbolic Notation (e.g., rwxr-xr--) to Octal\n");
        printf("3. Convert Octal Notation (e.g., 754) to Symbolic\n");
        printf("4. Exit Simulation\n");
        printf("--------------------------------------------------\n");
        printf("Enter your choice (1-4): ");
        
        if (scanf("%d", &choice) != 1) {
            printf("Invalid input. Please enter a number.\n");
            clear_input_buffer();
            continue;
        }

        switch (choice) {
            case 1:
                display_permissions_info();
                break;

            case 2:
                printf("\nEnter a 9-character symbolic string (e.g., rwxr-xr--): ");
                scanf("%9s", symbolic_input);
                if (strlen(symbolic_input) != 9) {
                    printf("Error: Symbolic notation must be exactly 9 characters long.\n");
                } else {
                    parse_symbolic_notation(symbolic_input);
                }
                break;

            case 3:
                printf("\nEnter a 3-digit octal permission value (e.g., 754): ");
                if (scanf("%d", &octal_input) != 1) {
                    printf("Error: Please enter a valid number.\n");
                    clear_input_buffer();
                } else if (octal_input < 0 || octal_input > 777) {
                    printf("Error: Octal value must be within the range 000 to 777.\n");
                } else {
                    parse_octal_notation(octal_input);
                }
                break;

            case 4:
                printf("Exiting Linux Permission Simulation Dashboard. Goodbye!\n");
                exit(0);

            default:
                printf("Invalid selection! Please choose an option from 1 to 4.\n");
        }
    }
    return 0;
}

void display_permissions_info() {
    printf("\n--- LINUX PERMISSION ARCHITECTURE THEORY ---\n\n");
    printf("1. USER TYPES:\n");
    printf("   - Owner (u)  : The user account that created the file asset.\n");
    printf("   - Group (g)  : A collection of users sharing identical execution access.\n");
    printf("   - Others (o) : Every other system entity profile (the rest of the world).\n\n");
    printf("2. ACCESS TYPE WEIGHTS:\n");
    printf("   - Read (r)    : Binary weight = 4. Allows viewing files / listing directories.\n");
    printf("   - Write (w)   : Binary weight = 2. Allows altering files / adding files to directories.\n");
    printf("   - Execute (x) : Binary weight = 1. Allows running files / passing through directories.\n");
}

void parse_symbolic_notation(const char *symbolic) {
    int octal_digits[3] = {0, 0, 0};
    const char *labels[3] = {"Owner (u)", "Group (g)", "Others (o)"};

    printf("\n--- Decoding Symbolic String: [%s] ---\n", symbolic);

    for (int i = 0; i < 3; i++) {
        int base_idx = i * 3;
        printf("%s Access: ", labels[i]);
        
        if (symbolic[base_idx] == 'r') { octal_digits[i] += 4; printf("Read "); }
        if (symbolic[base_idx + 1] == 'w') { octal_digits[i] += 2; printf("Write "); }
        if (symbolic[base_idx + 2] == 'x') { octal_digits[i] += 1; printf("Execute"); }
        
        if (octal_digits[i] == 0) {
            printf("No Permissions (---)");
        }
        printf("\n");
    }

    printf("Calculated Octal Representation: %d%d%d\n", octal_digits[0], octal_digits[1], octal_digits[2]);
}

void parse_octal_notation(int octal) {
    int digits[3];
    char symbolic[10] = "---------";
    const char *labels[3] = {"Owner (u)", "Group (g)", "Others (o)"};

    digits[2] = octal % 10;
    digits[1] = (octal / 10) % 10;
    digits[0] = (octal / 100) % 10;

    for (int i = 0; i < 3; i++) {
        if (digits[i] > 7) {
            printf("Error: Invalid octal digit '%d' detected. System base ranges from 0-7.\n", digits[i]);
            return;
        }
    }

    printf("\n--- Decoding Octal Code: [%d%d%d] ---\n", digits[0], digits[1], digits[2]);

    for (int i = 0; i < 3; i++) {
        int val = digits[i];
        int base_idx = i * 3;
        printf("%s Access Profile: ", labels[i]);

        if (val & 4) { symbolic[base_idx] = 'r'; printf("Read "); }
        if (val & 2) { symbolic[base_idx + 1] = 'w'; printf("Write "); }
        if (val & 1) { symbolic[base_idx + 2] = 'x'; printf("Execute"); }
        
        if (val == 0) {
            printf("No Access Rules Specified");
        }
        printf("\n");
    }

    printf("Generated Symbolic String: %s\n", symbolic);
}

void clear_input_buffer() {
    int c;
    while ((c = getchar()) != '\n' && c != EOF);
}
