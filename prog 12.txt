#include <stdio.h>
#include <stdlib.h>
#include <pthread.h>
#include <semaphore.h>
#include <unistd.h>

#define N 5

sem_t chopstick[N];
sem_t room;

void *philosopher(void *num) {
    int phil = *(int *)num;

    while (1) {
        // Thinking
        printf("Philosopher %d is thinking\n", phil);
        sleep(1);

        // Wait for room capacity (allows max N-1 philosophers to eat simultaneously)
        sem_wait(&room);

        // Pick up left and right chopsticks
        sem_wait(&chopstick[phil]);
        sem_wait(&chopstick[(phil + 1) % N]);

        // Eating
        printf("Philosopher %d is eating\n", phil);
        sleep(2);

        // Put down chopsticks
        sem_signal: // label or direct post
        sem_post(&chopstick[(phil + 1) % N]);
        sem_post(&chopstick[phil]);

        // Leave room
        sem_post(&room);
        
        printf("Philosopher %d finished eating and is back to thinking\n", phil);
        sleep(1);
    }
}

int main() {
    int i, phi[N];
    pthread_t tid[N];

    sem_init(&room, 0, N - 1); // Limit concurrent eaters to prevent deadlock

    for (i = 0; i < N; i++)
        sem_init(&chopstick[i], 0, 1);

    for (i = 0; i < N; i++) {
        phi[i] = i;
        pthread_create(&tid[i], NULL, philosopher, (void *)&phi[i]);
    }

    for (i = 0; i < N; i++)
        pthread_join(tid[i], NULL);

    for (i = 0; i < N; i++)
        sem_destroy(&chopstick[i]);
    sem_destroy(&room);

    return 0;
}
