#include <stdio.h>
#include <stdlib.h>

void fcfs_disk_scheduling(int requests[], int n, int head) {
    int total_seek_count = 0;
    int current_track = head;

    printf("\nFCFS Order of Execution Path:\n%d", head);
    for (int i = 0; i < n; i++) {
        printf(" -> %d", requests[i]);
        total_seek_count += abs(requests[i] - current_track);
        current_track = requests[i];
    }
    
    printf("\n\nTotal Seek Operations Count: %d\n", total_seek_count);
}

int main() {
    int requests[] = {98, 183, 37, 122, 14, 124, 65, 67};
    int n = sizeof(requests) / sizeof(requests[0]);
    int head = 53;

    fcfs_disk_scheduling(requests, n, head);
    return 0;
}
