#include <stdio.h>

struct Employee {
    int id;
    char name[30];
    float salary;
};

int main() {
    struct Employee e;
    FILE *fp;
    int n;

    fp = fopen("employee.dat", "wb+");

    if (fp == NULL) {
        printf("File cannot be opened.\n");
        return 1;
    }

    printf("Enter number of employees: ");
    scanf("%d", &n);

    for (int i = 0; i < n; i++) {
        printf("\nEnter Employee ID: ");
        scanf("%d", &e.id);

        printf("Enter Employee Name: ");
        scanf("%s", e.name);

        printf("Enter Salary: ");
        scanf("%f", &e.salary);

        fwrite(&e, sizeof(e), 1, fp);
    }

    printf("\nEnter record number to access: ");
    scanf("%d", &n);

    fseek(fp, (n - 1) * sizeof(e), SEEK_SET);

    if (fread(&e, sizeof(e), 1, fp) == 1) {
        printf("\nEmployee Details\n");
        printf("ID     : %d\n", e.id);
        printf("Name   : %s\n", e.name);
        printf("Salary : %.2f\n", e.salary);
    } else {
        printf("Record not found.\n");
    }

    fclose(fp);

    return 0;
}
