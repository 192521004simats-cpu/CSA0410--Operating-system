#include <stdio.h>
#include <pthread.h>
#include <unistd.h>

#define NUM_THREADS 2
#define ITERATIONS 100000

int shared_counter = 0;
pthread_mutex_t lock;

void* increment_counter(void* thread_id) {
    long tid = (long)thread_id;

    pthread_mutex_lock(&lock);
    
    printf("Thread %ld entered critical section.\n", tid);
    for (int i = 0; i < ITERATIONS; i++) {
        shared_counter++;
    }
    printf("Thread %ld exiting critical section. Current Counter: %d\n", tid, shared_counter);
    pthread_mutex_unlock(&lock);
    
    pthread_exit(NULL);
}

int main() {
    pthread_t threads[NUM_THREADS];
    
    if (pthread_mutex_init(&lock, NULL) != 0) {
        printf("Mutex initialization failed.\n");
        return 1;
    }

    for (long i = 0; i < NUM_THREADS; i++) {
        pthread_create(&threads[i], NULL, increment_counter, (void*)i);
    }

    for (int i = 0; i < NUM_THREADS; i++) {
        pthread_join(threads[i], NULL);
    }

    pthread_mutex_destroy(&lock);

    printf("Final Shared Counter Value (Expected %d): %d\n", NUM_THREADS * ITERATIONS, shared_counter);
    return 0;
}
